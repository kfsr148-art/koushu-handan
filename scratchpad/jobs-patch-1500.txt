const fs = require('fs'); const root = process.argv[2];
function edit(file, fn) {
  const p = root + '/' + file; const raw = fs.readFileSync(p, 'utf8'); const crlf = raw.indexOf('\r\n') >= 0;
  let s = raw.split('\r\n').join('\n');
  const rep = (a, b) => { if (s.split(a).length !== 2) throw new Error(file + ' uniq ' + a.slice(0, 70)); s = s.replace(a, () => b); };
  fn(rep); fs.writeFileSync(p, crlf ? s.split('\n').join('\r\n') : s);
}
