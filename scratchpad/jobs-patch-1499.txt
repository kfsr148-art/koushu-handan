// 使い方: node patch1499.js <root>   root の advice-tone-core.js に、方言と人物の読み直しの直しを当てる（y1116-6）
const fs = require('fs'); const root = process.argv[2];
const p = root + '/advice-tone-core.js';
const raw = fs.readFileSync(p, 'utf8'); const crlf = raw.indexOf('\r\n') >= 0;
let s = raw.split('\r\n').join('\n');
const rep = (a, b) => { if (s.split(a).length !== 2) throw new Error('uniq ' + a.slice(0, 60)); s = s.replace(a, () => b); };

// 1. 人物の表（呼びかけ・口癖）
rep("c: 'あなた', k: 'いけずやなかですわ'", "c: 'あなた', k: 'あらまあ'");              // 博多の人が京ことば（いけず）を言っていた
rep("c: 'にいちゃん', k: 'なんのだや'", "c: 'にいちゃん', k: 'なのやで'");             // 意味の通らない口癖
rep("c: 'おいやすさん', k: 'なのだどす'", "c: 'おきゃくはん', k: 'なのどす'");          // 「おいやす」は呼びかけではなく挨拶
rep("c: 'おいやす', k: 'ほんまに'", "c: 'おきゃくはん', k: 'ほんまに'");
rep("c: 'あなた', k: 'いらっしゃいませ'", "c: 'あなた', k: '聞いてくださいな'");        // 助言の頭で「いらっしゃいませ」は変
// 2. つなぎ（博多）
rep("'あとは、', 'じゃけん、'", "'あとは、', 'やけん、'");
// 3. 方言ごとの丁寧な形
rep("'ましょう': 'ましょうえ', 'ませ': 'ませ' }[e] || e + 'え'", "'ましょう': 'ましょうか', 'ませ': 'ませ' }[e] || e + 'え'");
// 4. 語尾の三通り
rep("po: ['っちゃ', 'ちゃが'] }", "po: ['っちゃ', 'とね'] }");
rep("pl: ['たい', 'けん'], po: ['ばい', 'たい'] }", "pl: ['たい', 'やけん'], po: ['ばい', 'たい'] }");
rep("po: ['なぁ', 'で'] }", "po: ['なぁ', 'わな'] }");
rep("pl: ['わぁ', 'おす'], po: ['なぁ', 'わぁ'] }", "pl: ['わぁ', 'どす'], po: ['なぁ', 'わぁ'] }");
rep("aomori: { pl: ['だべ', 'んだ'], po: ['だべ', 'んだ'] }", "aomori: { pl: ['だべ', 'んだ'], po: ['だべ', 'だ'] }");
rep("po: ['ぞ', 'ぞよ'] }", "po: ['ぞ', 'わい'] }");
// 5. 名詞・のだ に語尾を付ける所
rep("  if (w.charAt(0) === 'だ' && /だ$/.test(core)) { return core + w.slice(1); }\n",
"  if (w.charAt(0) === 'だ' && /だ$/.test(core)) { return core + w.slice(1); }\n" +
"  if (dk === 'yoji' && w === 'の' && /のだ$/.test(core)) { return core + 'ー'; }   // 「のだの」は変（y1116-6）\n" +
"  if (dk === 'hakata' && /のだ$/.test(core) && w !== 'ばい') { return w === 'たい' ? core.slice(0, -2) + 'とたい' : core + 'けん'; }   // 「のだたい」は変。「見るとたい」が博多\n" +
"  if ((dk === 'osaka' || dk === 'kyoto') && /のだ$/.test(core)) { return core.slice(0, -1) + w; }\n" +
"  if ((dk === 'hakata' || dk === 'osaka' || dk === 'kyoto') && /だ$/.test(core) && !/のだ$/.test(core)) { return core.slice(0, -1) + w; }   // 「二兎だたい」「一向聴だやん」「だおす」は変。名詞に「だ」を付けない\n");
// 6. 武士
rep("    if (dk === 'bushi' && v === 1) { if (/のだ$/.test(core)) { return core.replace(/のだ$/, 'のじゃ'); } if (/だ$/.test(core)) { return core.replace(/だ$/, 'じゃ'); } }\n",
"    if (dk === 'bushi' && v === 1) { if (/のだ$/.test(core)) { return core.replace(/のだ$/, 'のじゃ'); } if (/だ$/.test(core)) { return core.replace(/だ$/, 'じゃ'); } if (!/にゃ$/.test(core)) { return core + 'のじゃ'; } }   // 「見るじゃ」は変。「見るのじゃ」\n" +
"    if (dk === 'bushi' && v === 2) { if (/のだ$/.test(core)) { return core.replace(/のだ$/, 'のでござるぞ'); } if (/だ$/.test(core)) { return core.replace(/だ$/, 'でござるぞ'); } }   // 「なのだでござるぞ」は変\n");
rep("  if (p === 'ましょう') { return core + (v === 1 ? 'ね' : 'よ'); }   // 「ましょう」に方言の語を足すと不自然になる所（ましょうんだ 等）は、ね／よ にする\n",
"  if (p === 'ましょう') { return core + ADV_MASHO[dk][v - 1]; }   // 「ましょう」の後ろは方言ごとに決める（標準語の「ね」を残さない・y1116-6）\n");
rep("    if (dk === 'bushi' && v === 1) { const r = {",
"    if (dk === 'bushi' && v === 2) { const r2 = { 'ございません': 'ござらぬぞ', 'ございます': 'ござるぞ', 'おりません': 'おらぬぞ', 'おります': 'おるぞ', 'いたしません': 'いたさぬぞ', 'いたします': 'いたすぞ', 'ません': 'ませぬぞ' }[p]; if (r2) { return core.slice(0, -p.length) + r2; } }\n    if (dk === 'bushi' && v === 1) { const r = {");
rep("function advDialSentenceV(core, dk, polite, v) {",
"const ADV_MASHO = { miyazaki: ['や', 'とよ'], hakata: ['たい', 'や'], osaka: ['な', 'やないか'], kyoto: ['なぁ', 'やおへんか'], aomori: ['な', 'や'], okinawa: ['ねー', 'やー'], bushi: ['ぞ', 'ぞ'], yoji: ['ね', 'よ'] };\n" +
"function advBushiFem(t) {   // 姫君・奥方様は、武士の語尾（ござる・ぞ・じゃ）を女の言い方に直す（y1116-6）\n" +
"  return t.replace(/[^。！？]+[。！？]/g, function (st) {\n" +
"    const end = st.slice(-1); let b = st.slice(0, -1);\n" +
"    [[/(で|に)ござる(ぞ)?$/, 'でございます'], [/ござりまする$/, 'ございます'], [/ござらぬ(ぞ)?$/, 'ございません'], [/ござる(ぞ)?$/, 'ございます'], [/まする$/, 'ますのよ'], [/まするぞ$/, 'ますのよ'], [/ませぬぞ?$/, 'ません'], [/ますぞ$/, 'ますわ'], [/ますわい$/, 'ますことよ'], [/ですぞ$/, 'ですわ'], [/ですわい$/, 'ですことよ'], [/ましょうぞ$/, 'ましょう'], [/なされよ$/, 'なさいませ'], [/おるぞ$/, 'おります'], [/おらぬぞ$/, 'おりません'], [/いたすぞ$/, 'いたします'], [/のじゃ$/, 'のよ'], [/じゃ$/, 'わ'], [/ぞ$/, 'わ']].some(function (r) { if (r[0].test(b)) { b = b.replace(r[0], r[1]); return true; } return false; });\n" +
"    return b + end;\n  });\n}\n" +
"function advDialSentenceV(core, dk, polite, v) {");
// 7. 組み立て：マダムの「ございます」をそろえる／姫君・奥方様の女言葉
rep("      const base = (v >= 0) ? tk.t.replace(/、(?:ウィ|モナミ)(?=[。！？])/g, '') : tk.t;   // 方言の世界の人物は、外国語の呼びかけを言わない\n",
"      let base = (v >= 0) ? tk.t.replace(/、(?:ウィ|モナミ)(?=[。！？])/g, '') : tk.t;   // 方言の世界の人物は、外国語の呼びかけを言わない\n" +
"      if (v >= 0 && tone === 'french') { base = base.replace(/ようございます/g, 'いいです').replace(/でございません/g, 'ません').replace(/でございます/g, 'です').replace(/ございません/g, 'ません').replace(/ございます/g, 'です'); }   // マダムは「です」でそろえる（「ございます」と混ぜない・y1116-6）\n");
rep("      if (v >= 0) { const q = advDialAll(base, dial, polite, v, ends4); x = q.text; vv = q.v; }\n",
"      if (v >= 0) { const q = advDialAll(base, dial, polite, v, ends4); x = q.text; vv = q.v; if (dial === 'bushi' && (tone === 'lady' || tone === 'french')) { x = advBushiFem(x); } }\n");
fs.writeFileSync(p, crlf ? s.split('\n').join('\r\n') : s); console.log('patched', root);
